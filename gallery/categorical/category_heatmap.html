Category Heatmap#

A heatmap showing the relationship between two categorical variables.
This example uses the earthquakes dataset from hvsampledata to count occurrences by magnitude and depth class.

import hvplot.pandas  # noqa
import hvsampledata
import numpy as np

df = hvsampledata.earthquakes('pandas')
grouped = df.groupby(['depth_class', 'mag_class'], observed=True).size().reset_index(name='count')

grouped.hvplot.heatmap(
    x='mag_class',
    y='depth_class',
    C='count',
    reduce_function=np.mean,
    logz=True,
    title='Category Heatmap (Bokeh)',
    xlabel='Magnitude Class',
    ylabel='Depth Class',
)

import hvplot.pandas  # noqa
import hvsampledata
import numpy as np

hvplot.extension('matplotlib')

df = hvsampledata.earthquakes('pandas')

grouped = df.groupby(['depth_class', 'mag_class'], observed=True).size().reset_index(name='count')

grouped.hvplot.heatmap(
    x='mag_class',
    y='depth_class',
    C='count',
    reduce_function=np.mean,
    logz=True,
    title='Category Heatmap (Matplotlib)',
    xlabel='Magnitude Class',
    ylabel='Depth Class',
)
